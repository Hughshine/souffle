 SUBROUTINE A_1_negation_subproof
  QUERY
   IF (ARGUMENT(0),ARGUMENT(1),UNDEF,UNDEF) IN B
    RETURN (NUMBER(1))
  END QUERY
  QUERY
   IF (NOT (ARGUMENT(0),ARGUMENT(1),UNDEF,UNDEF) IN B)
    RETURN (NUMBER(0))
  END QUERY
  QUERY
   IF (ARGUMENT(1),UNDEF,UNDEF) IN C
    RETURN (NUMBER(0))
  END QUERY
  QUERY
   IF (NOT (ARGUMENT(1),UNDEF,UNDEF) IN C)
    RETURN (NUMBER(1))
  END QUERY
 END SUBROUTINE
 SUBROUTINE A_1_subproof

   FOR t0 IN B
    IF (NOT ISEMPTY(B))
     IF (t0.3 < ARGUMENT(1))
      IF (t0.0 = ARGUMENT(0))

        RETURN (t0.0, t0.1, t0.2, t0.3, t0.1, UNDEF, UNDEF, t0.0, ARGUMENT(0), t0.3, ARGUMENT(1))
